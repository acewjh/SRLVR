# opsd-slime: independent Qwen3-1.7B math OPSD harness on Ascend 910B2 (8x64GB HBM)
# Python 3.10, CANN 8.5.1, driver 25.5.1.
#
# Reproduces the reference slime NPU env (validated on 910B2 + CANN 8.5.1, see
# cua/slime-geo3k-rl-npu/requirements.txt). CUDA-only wheels (nvidia-*, CUDA triton)
# are deliberately NOT installed.
#
# Install:
#   conda create -n slime-npu python=3.10 -y && conda activate slime-npu
#   pip install -r configs/requirements.txt
#   pip install -e frameworks/Megatron-LM      --no-deps
#   pip install -e frameworks/Megatron-Bridge  --no-deps
#   pip install -e frameworks/mbridge          --no-deps
#   pip install -e frameworks/MindSpeed        --no-deps
#   pip install -e frameworks/slime-ascend     --no-deps
#   pip install -e frameworks/sglang/python    --no-deps   # MUST be last
#
# Why torch 2.1.0: torch-npu 2.1.0.post17 pins torch==2.1.0 (validated combo with
# the vendored sglang fork; torch>=2.2 breaks its torch._custom_op usage).
# Why transformers 4.51.3 (NOT the 4.57.6 of the reference list): transformers
# >= 4.56 calls torch.utils._pytree.register_pytree_node, which only exists in
# torch >= 2.4, and therefore cannot import under the pinned torch 2.1.0.
# 4.51.3 is the newest torch-2.1-compatible line that ships Qwen3 model classes
# (needed by the frozen HF teacher); Megatron-Bridge logs "skipping model family"
# notices for newer architectures, which is expected and harmless.
